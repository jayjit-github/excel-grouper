package com.excel.grouping;

import java.io.File;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

public class Main {

    public static void main(String[] args) {
        String inputFilePath;
        boolean mergeToOneFile = false;

        // Case 1: Running from command-line
        if (args.length >= 1) {
            inputFilePath = args[0];
            if (args.length > 1 && args[1].equalsIgnoreCase("--merge")) {
                mergeToOneFile = true;
            }
        }
        // Case 2: Running by directly providing input (for testing)
        else {
            inputFilePath = "/Users/jayjitmandal/codebase/excel-grouper/history-data.csv"; // Change default for testing
            mergeToOneFile = false;
        }

        File inputFile = new File(inputFilePath);
        if (!inputFile.exists() || !inputFile.isFile()) {
            System.err.println("Invalid input file path: " + inputFilePath);
            System.exit(1);
        }

        List<NiftyRecord> records;
        try {
            if (inputFilePath.endsWith(".xlsx")) {
                records = ExcelUtils.readExcelFile(inputFilePath);
            } else if (inputFilePath.endsWith(".csv")) {
                records = ExcelUtils.readCsvFile(inputFilePath);
            } else {
                System.err.println("Unsupported file format (must be .csv or .xlsx)");
                System.exit(1);
                return;
            }
        } catch (IOException e) {
            System.err.println("Error reading input file: " + e.getMessage());
            return;
        }

        if (records.isEmpty()) {
            System.err.println("No data found in input file.");
            System.exit(1);
        }

        // Group by week and month

        // List<NiftyRecord> records = new ArrayList<>();
        // records.add(new NiftyRecord("2024-01-01", 21517.35, 21661.10, 21677.00, 21500.35));
        // records.add(new NiftyRecord("2024-01-02", 21618.70, 21529.30, 21641.85, 21448.65));
        // records.add(new NiftyRecord("2024-01-02", 21571.95, 21647.25, 21851.50, 21550.45));
        // records.add(new NiftyRecord("2024-01-04", 21453.95, 21185.25, 21482.35, 21137.20));
        // records.add(new NiftyRecord("2024-01-05", 21725.70, 21487.25, 21741.35, 21448.85));
        // records.add(new NiftyRecord("2024-01-06", 21930.50, 22045.05, 22053.30, 21860.15));
        // records.add(new NiftyRecord("2024-01-07", 21840.05, 21578.15, 21870.85, 21530.20));
        // records.add(new NiftyRecord("2024-01-08", 22055.05, 22248.85, 22249.40, 21997.95));
        // records.add(new NiftyRecord("2024-01-09", 21951.15, 22214.10, 22229.15, 21860.65));

        Map<String, List<NiftyRecord>> weeklyGrouped = NiftyDataGrouper.groupByWeek(records);
        Map<String, List<NiftyRecord>> monthlyGrouped = NiftyDataGrouper.groupByMonth(records);

        String baseName = inputFile.getName().replaceAll("\\.csv|\\.xlsx", "");
        String outputDir = inputFile.getParent();
        if (outputDir == null) {
            outputDir = "."; // Current directory fallback
        }

        if (mergeToOneFile) {
            String mergedFilePath = outputDir + File.separator + baseName + "_weekly_monthly.xlsx";
            ExcelUtils.writeWeeklyAndMonthlyToOneFile(mergedFilePath, weeklyGrouped, monthlyGrouped);
            System.out.println("✅ Merged weekly and monthly file created at: " + mergedFilePath);
        } else {
            String weeklyOutput = outputDir + File.separator + baseName + "_weekly.xlsx";
            String monthlyOutput = outputDir + File.separator + baseName + "_monthly.xlsx";

            ExcelUtils.writeGroupedRecordsToExcel(weeklyOutput, weeklyGrouped);
            ExcelUtils.writeGroupedRecordsToExcel(monthlyOutput, monthlyGrouped);

            System.out.println("✅ Weekly file created at: " + weeklyOutput);
            System.out.println("✅ Monthly file created at: " + monthlyOutput);
        }
    }
}
